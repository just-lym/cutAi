## config.py - 全局配置

使用 pydantic-settings，支持 .env 文件和环境变量：

```python
from pathlib import Path
from pydantic_settings import BaseSettings

class Settings(BaseSettings):
    database_url: str = "postgresql+asyncpg://aicut:aicut@localhost:5432/aicut"
    redis_url: str = "redis://localhost:6379/0"

    dashscope_api_key: str = ""
    video_gen_provider: str = "runway"  # runway | pika | keling
    runway_api_key: str = ""
    pika_api_key: str = ""
    keling_api_key: str = ""

    monthly_budget_yuan: float = 100.0
    daily_budget_yuan: float = 10.0

    data_root: Path = Path("/data/aicut")

    gpu_render_concurrency: int = 1
    ffmpeg_hwaccel: str = "cuda"

    model_config = {"env_file": ".env", "env_file_encoding": "utf-8"}

    @property
    def projects_root(self) -> Path:
        return self.data_root / "projects"

settings = Settings()
from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker, create_async_engine
from sqlalchemy.orm import DeclarativeBase, Session, sessionmaker
from sqlalchemy import create_engine

engine = create_async_engine(settings.database_url, echo=False)
async_session = async_sessionmaker(engine, class_=AsyncSession, expire_on_commit=False)

# Sync 引擎：URL 替换 asyncpg -> psycopg2
sync_db_url = settings.database_url.replace("postgresql+asyncpg", "postgresql+psycopg2")
sync_engine = create_engine(sync_db_url, echo=False)
SyncSession = sessionmaker(sync_engine, class_=Session, expire_on_commit=False)

class Base(DeclarativeBase):
    pass

async def get_db() -> AsyncSession:
    async with async_session() as session:
        yield session

@contextmanager
def get_sync_session():
    """Sync session for LangChain tools."""
    session = SyncSession()
    try:
        yield session
    finally:
        session.close()
```

## WebSocket 事件管理

```python
class ConnectionManager:
    def __init__(self):
        self.connections: dict[str, list[WebSocket]] = defaultdict(list)

    async def connect(self, project_id: str, ws: WebSocket):
        await ws.accept()
        self.connections[project_id].append(ws)

    def disconnect(self, project_id: str, ws: WebSocket):
        self.connections[project_id].remove(ws)
```

## arq Worker
轻量异步任务队列，处理 GPU 密集型任务

```python
class WorkerSettings:
    redis_settings = RedisSettings.from_dsn(settings.redis_url)
    functions = [
        task_media_probe,           # FFprobe 媒体信息
        task_proxy_generation,      # GPU 代理视频
        task_thumbnail_generation,  # 缩略图序列
        task_audio_extraction,      # 音频提取
        task_preview_render,        # GPU 预览渲染
        task_final_render,          # GPU 最终导出
    ]
    
    max_jobs = settings.gpu_render_concurrency * 4  # GPU + IO 并行
    job_timeout = 3600
```
